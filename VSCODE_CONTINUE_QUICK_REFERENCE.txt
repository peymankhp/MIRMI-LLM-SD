📋 MANUAL SETUP STEPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Step 1: Install VSCode
    sudo snap install code --classic

  Step 2: Pull recommended models
    docker exec ollama ollama pull qwen2.5-coder:7b
    docker exec ollama ollama pull nomic-embed-text

  Step 3: Open VSCode
    code

  Step 4: Install Continue extension
    • Press Ctrl+Shift+X
    • Search "Continue"
    • Click Install

  Step 5: Configure Continue
    • Press Ctrl+Shift+P
    • Type "Continue: Open Config"
    • Paste configuration (see full guide)

🔧 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Problem: "Failed to connect to Ollama"
  Solution:
    docker restart ollama
    curl http://localhost:11434/api/tags

  Problem: "Model not found"
  Solution:
    docker exec ollama ollama list
    docker exec ollama ollama pull qwen2.5-coder:7b

  Problem: Slow responses
  Solution:
    • Switch to llama3.1:8b (faster model)
    • Check GPU: nvidia-smi
    • Close other GPU applications

  Problem: Autocomplete not working
  Solution:
    • Wait 2-3 seconds after typing
    • Check tabAutocompleteModel in config
    • Try pressing Tab manually
