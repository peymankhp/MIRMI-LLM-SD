📝 PARAMETER EXPLANATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  steps:            Number of generation steps (20-50)
                    • 20 = Fast (5-8 seconds)
                    • 30 = Balanced (10-15 seconds)
                    • 50 = High quality (20-30 seconds)

  cfg_scale:        How closely to follow prompt (7-12)
                    • 7 = More creative
                    • 10 = Balanced
                    • 12 = Very literal

  sampler_name:     Algorithm used for generation
                    • "Euler a" = Fast
                    • "DPM++ 2M Karras" = Best quality
                    • "DDIM" = Consistent results

  negative_prompt:  What to avoid in the image
                    Common: "blurry, low quality, distorted"

💡 TIPS FOR BETTER IMAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ✅ Be specific and descriptive
     Good: "a red sports car on a mountain road, sunset, photorealistic"
     Bad: "car"

  ✅ Add quality keywords
     "highly detailed", "8k", "professional", "sharp focus"

  ✅ Specify style
     "photorealistic", "oil painting", "digital art", "anime style"

  ✅ Include lighting
     "studio lighting", "golden hour", "dramatic lighting"

  ✅ Use negative prompts
     "blurry, low quality, distorted, ugly, watermark"

🔍 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Issue: "Connection error" or "API not responding"
  
  Solution:
    1. Check if Stable Diffusion is running:
       docker ps | grep automatic1111
    
    2. Restart if needed:
       docker restart automatic1111
    
    3. Wait 30 seconds and try again

  Issue: "Out of memory" error
  
  Solution:
    1. Reduce image size to 512x512
    2. Reduce steps to 20
    3. Check GPU usage: nvidia-smi

  Issue: Images are blurry or low quality
  
  Solution:
    1. Increase steps to 30-50
    2. Add quality keywords to prompt
    3. Use negative prompt
    4. Try different sampler
