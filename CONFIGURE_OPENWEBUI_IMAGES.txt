🔍 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Problem: "Connection error" when generating image
  
  Solution:
    1. Check if Stable Diffusion is running:
       docker ps | grep automatic1111
    
    2. If not running, start it:
       docker start automatic1111
    
    3. Wait 30 seconds and try again

  Problem: Image generation is very slow
  
  Solution:
    1. Reduce Image Size to 512x512
    2. Reduce Steps to 20
    3. Check GPU usage: nvidia-smi

  Problem: Images are low quality
  
  Solution:
    1. Increase Steps to 40-50
    2. Add quality keywords to your prompt:
       "highly detailed, 8k, photorealistic"
    3. Use negative prompt in Additional Parameters

💡 TIPS FOR BETTER IMAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ✓ Be specific and descriptive in prompts
  ✓ Add quality keywords: "highly detailed", "8k", "professional"
  ✓ Specify style: "photorealistic", "oil painting", "anime style"
  ✓ Include lighting: "studio lighting", "golden hour", "dramatic"
  ✓ Use negative prompts to avoid unwanted elements
